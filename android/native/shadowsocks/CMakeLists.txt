enable_language(ASM)
set(DEPS "${CMAKE_CURRENT_LIST_DIR}/../../../.build-tools/ss-deps")
set(SS "${DEPS}/shadowsocks/src")
if(NOT EXISTS "${SS}/server.c")
    message(FATAL_ERROR "Run python android/prepare-shadowsocks.py first")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(PCRE_BUILD_PCRECPP OFF CACHE BOOL "" FORCE)
set(PCRE_BUILD_PCREGREP OFF CACHE BOOL "" FORCE)
set(PCRE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE_SUPPORT_JIT OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/pcre" pcre)
set(CARES_STATIC ON CACHE BOOL "" FORCE)
set(CARES_SHARED OFF CACHE BOOL "" FORCE)
set(CARES_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CARES_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
# Bionic implements pthreads in libc; FindThreads tests pthread_cancel, which
# Android deliberately does not implement.
set(CMAKE_HAVE_LIBC_PTHREAD 1 CACHE INTERNAL "")
add_subdirectory("${DEPS}/cares" cares)
add_library(ss_ev STATIC "${DEPS}/libev/ev.c")
target_compile_definitions(ss_ev PRIVATE EV_STANDALONE=1)
target_include_directories(ss_ev PUBLIC "${DEPS}/libev")
file(GLOB CORK_CORE "${DEPS}/libcork/src/libcork/core/*.c")
file(GLOB CORK_DS "${DEPS}/libcork/src/libcork/ds/*.c")
file(GLOB CORK_POSIX "${DEPS}/libcork/src/libcork/posix/*.c")
list(FILTER CORK_POSIX EXCLUDE REGEX "mingw\\.c$")
add_library(ss_cork STATIC ${CORK_CORE} ${CORK_DS} ${CORK_POSIX} "${DEPS}/libcork/src/libcork/pthreads/thread.c")
target_include_directories(ss_cork PUBLIC "${DEPS}/libcork/include")
target_compile_definitions(ss_cork PUBLIC CORK_API=CORK_LOCAL)
file(GLOB_RECURSE IPSET_SOURCES "${DEPS}/libipset/src/libipset/*.c")
add_library(ss_ipset STATIC ${IPSET_SOURCES})
target_include_directories(ss_ipset PUBLIC "${DEPS}/libipset/include")
target_link_libraries(ss_ipset PUBLIC ss_cork)
add_library(ss_bloom STATIC "${DEPS}/libbloom/bloom.c" "${DEPS}/libbloom/murmur2/MurmurHash2.c")
target_include_directories(ss_bloom PUBLIC "${DEPS}/libbloom" "${DEPS}/libbloom/murmur2")
file(GLOB CRYPTO_C "${SS}/vendor/aws-lc/crypto/*.c")
file(GLOB_RECURSE CRYPTO_ASM "${SS}/vendor/aws-lc/asm/*.S")
add_library(ss_crypto STATIC ${CRYPTO_C} ${CRYPTO_ASM} "${SS}/vendor/aws-lc-glue.c")
target_include_directories(ss_crypto PUBLIC "${SS}/vendor" "${SS}/vendor/aws-lc/include" PRIVATE "${SS}/vendor/aws-lc")
target_compile_definitions(ss_crypto PRIVATE NDEBUG)
set(COMMON utils.c jconf.c json.c udprelay.c cache.c netutils.c crypto.c aead.c stream.c md5.c hkdf.c libm_shim.c ppbloom.c base64.c plugin.c rule.c acl.c)
list(TRANSFORM COMMON PREPEND "${SS}/")
# Report readiness only after all requested listeners have been initialized.
file(READ "${SS}/server.c" ANDROID_SERVER_SOURCE)
string(REPLACE "    ev_run(loop, 0);" "    fprintf(stderr, \"IPM_SS_READY\\n\"); fflush(stderr);\n    ev_run(loop, 0);" ANDROID_SERVER_SOURCE "${ANDROID_SERVER_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/android_server.c" "${ANDROID_SERVER_SOURCE}")
add_executable(ss_server "${CMAKE_CURRENT_BINARY_DIR}/android_server.c" "${SS}/resolv.c" ${COMMON} android_compat.c ../android_lifecycle.cpp)
target_include_directories(ss_server PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${SS}" "${DEPS}/pcre" "${CMAKE_CURRENT_BINARY_DIR}/pcre" "${DEPS}/cares/include" "${CMAKE_CURRENT_BINARY_DIR}/cares")
target_compile_definitions(ss_server PRIVATE HAVE_CONFIG_H MODULE_REMOTE __android_log_print=ss_file_log)
target_compile_options(ss_server PRIVATE -Os -fno-strict-aliasing -ffunction-sections -fdata-sections -Wno-deprecated-declarations "$<$<COMPILE_LANGUAGE:C>:-include${CMAKE_CURRENT_LIST_DIR}/android_compat.h>")
target_link_libraries(ss_server PRIVATE ss_ev ss_crypto ss_ipset ss_bloom pcre c-ares::cares_static m)
target_link_options(ss_server PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
set_target_properties(ss_server PROPERTIES OUTPUT_NAME "libss_server.so" C_STANDARD 99)
