cmake_minimum_required(VERSION 3.18)
project(intraportmap_android C CXX)
add_compile_definitions(_GNU_SOURCE)
set(CMAKE_CXX_STANDARD 11)
set(EVENT__DISABLE_OPENSSL ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_THREAD_SUPPORT ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_TESTS ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_REGRESS ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_SAMPLES ON CACHE BOOL "" FORCE)
set(EVENT__DISABLE_BENCHMARK ON CACHE BOOL "" FORCE)
set(EVENT__LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
add_subdirectory("${LIBEVENT_SOURCE}" libevent)
file(GLOB IPM_SOURCES "${CMAKE_CURRENT_LIST_DIR}/../../src/*.cpp")
add_executable(intraportmap ${IPM_SOURCES} android_dns.cpp android_lifecycle.cpp)
target_include_directories(intraportmap PRIVATE "${LIBEVENT_SOURCE}/include" "${CMAKE_CURRENT_BINARY_DIR}/libevent/include")
target_link_libraries(intraportmap event_extra_static event_core_static)
target_compile_definitions(intraportmap PRIVATE _GNU_SOURCE evdns_getaddrinfo=ipm_android_getaddrinfo evdns_base_new=ipm_android_dns_base_new)
target_compile_options(intraportmap PRIVATE -Os -fno-exceptions -fno-rtti -ffunction-sections -fdata-sections)
target_link_options(intraportmap PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
set_target_properties(intraportmap PROPERTIES OUTPUT_NAME "libintraportmap.so")
add_subdirectory(shadowsocks)
